// ==========================================
// GOOGLE APPS SCRIPT (Code.gs)
// Mở Google Sheets -> Tiện ích mở rộng (Extensions) -> Apps Script -> Dán đoạn code này:
// ==========================================
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var doc = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = doc.getActiveSheet();
    
    // Nếu sheet trống, thêm dòng tiêu đề
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Thời gian", "Họ tên", "Email", "Số điện thoại", "Hạng vé", "Mã Check-in"]);
    }
    
    // Đếm số hàng để kiểm tra giới hạn 50 người
    var totalRows = sheet.getLastRow();
    if (totalRows > 50) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "full",
        message: "Sự kiện đã đủ 50 người đăng ký. Rất tiếc không thể tiếp nhận thêm!"
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = JSON.parse(e.postData.contents);
    var timestamp = new Date();
    var checkInCode = "AI2026-" + ("" + Math.random()).substring(2, 8);
    
    sheet.appendRow([
      timestamp,
      data.fullName,
      data.email,
      data.phone,
      data.ticketType,
      checkInCode
    ]);
    
    // Tự động gửi Email xác nhận qua GmailApp
    try {
      var subject = "[AI Workshop 2026] Xác nhận đăng ký thành công cho " + data.fullName;
      var body = "Xin chào " + data.fullName + ",\n\n"
        + "Cảm ơn bạn đã đăng ký tham dự sự kiện 'AI & Future Careers Workshop 2026'!\n"
        + "- Hạng vé: " + data.ticketType + "\n"
        + "- Mã Check-in: " + checkInCode + "\n\n"
        + "LƯU Ý QUAN TRỌNG:\n"
        + "Vui lòng mang theo Laptop cá nhân có cài đặt trình duyệt web và sạc đầy pin để tham gia phần thực hành 'Vibe Coding' cùng diễn giả.\n\n"
        + "Hẹn gặp lại bạn tại hội trường!\n"
        + "Ban Tổ Chức AI Workshop 2026.";
      GmailApp.sendEmail(data.email, subject, body);
    } catch (mailErr) {
      Logger.log("Mail error: " + mailErr);
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      checkInCode: checkInCode,
      message: "Đăng ký thành công và đã ghi nhận vào Google Sheets!"
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
