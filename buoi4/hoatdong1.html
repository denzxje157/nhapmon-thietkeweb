--secondary-color Sakura Coral (Điểm nhấn)
--gray-light Mức xám 1 (Thẻ / Nền phụ)
--gray-medium Mức xám 2 (Đường viền)
--gray-dark Mức xám 3 (Chữ thứ cấp)
--bg-color Màu nền toàn trang
